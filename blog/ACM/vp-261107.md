#include<bits/stdc++.h>
using namespace std;
#define ll long long
//必然单边全取
int n, mid; ll s, ls, rs;
double la, ra, a;
vector<ll> v;

void solve(){
    cin>>n;
    v.resize(n+1); v[0] = 1e9 +1;
    cin>>v[1]; s = v[1];
    for(int i = 2; i<=n; i++){
        cin>>v[i]; s += v[i];
        if(v[i]>v[i-1] && v[i-1]<v[i-2]){mid = i-1; ls = s;}
    }
    rs = s - ls + v[mid] + v[mid+1] + v[mid-1];
    la = (double)ls / (double)(mid+1);
    ra = (double)rs / (double)(n-mid+2);
    a = (double)s / (double)n;

    double ans = max(la, ra);
    ans = max(ans, a);
    cout<<fixed<<setprecision(16)<<ans<<endl;

    //end
    v.clear();
}

//
int main(){
    //speed up
    ios::sync_with_stdio(0);
    cin.tie(0);
    int T = 1; cin>>T;
    while(T--) solve();
}
